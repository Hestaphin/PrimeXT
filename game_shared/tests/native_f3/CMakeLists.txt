cmake_minimum_required(VERSION 3.16)
project(primext_f3_drop LANGUAGES CXX)
get_filename_component(PRIMEXT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
list(APPEND CMAKE_MODULE_PATH "${PRIMEXT_ROOT}/cmake")
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE RelWithDebInfo CACHE STRING "Build configuration" FORCE)
endif()
find_package(PhysX REQUIRED)
find_package(Threads REQUIRED)
add_executable(primext-f3-drop drop_capture.cpp "${PRIMEXT_ROOT}/game_shared/physics_asset_format.cpp")
target_compile_features(primext-f3-drop PRIVATE cxx_std_17)
target_include_directories(primext-f3-drop PRIVATE "${PRIMEXT_ROOT}/game_shared" "${PRIMEXT_ROOT}/server/physx")
target_link_libraries(primext-f3-drop PRIVATE PhysX::PhysX Threads::Threads)
